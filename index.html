trade-timers

countdown.py 
def time_remaining(session_close: datetime) -> str:
    # always-on-top desktop widget, refreshed every tick
    delta = session_close - datetime.utcnow()
    if delta.total_seconds() <= 0:
        return "SESSION CLOSED"

    h, rem = divmod(delta.seconds, 3600)
    m, s = divmod(rem, 60)
    return f"{h:02d}:{m:02d}:{s:02d}"

export function useExpiringLink(ttlMinutes: number) {
  const expiresAt = ref(Date.now() + ttlMinutes * 60_000);
  const isValid = computed(() => Date.now() < expiresAt.value);

  return { expiresAt, isValid };
}
